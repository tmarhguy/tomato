\begin{center}
    {\LARGE \textbf{MOSAIC32 ALU Control Map - Reference Sheet}} \\[0.3em]
    {\small 14-bit micro-instruction: \texttt{LUT[7:0]} + \texttt{ctrl[4:0]} + \texttt{Cin} \quad \textbullet \quad $\text{A}_{\text{eff}} = (\text{Asel} \cdot A) \oplus \text{Ainv}$ \quad $\text{B}_{\text{eff}} = (\text{Bsel} \cdot B) \oplus \text{Binv}$} \\
    {\small \texttt{csel=0} $\rightarrow$ Arithmetic Mode (carry chain) \quad \textbullet \quad \texttt{csel=1} $\rightarrow$ Logic Mode (C bus)}
\end{center}
